# Raspberry Pi Boot Configuration for WebbOS
# This file is read by the GPU firmware at boot time
# Documentation: https://www.raspberrypi.com/documentation/computers/config_txt.html

# ============================================================================
# Architecture Configuration
# ============================================================================

# Enable 64-bit ARM mode (required for kernel8.img)
# When enabled, the bootloader loads kernel8.img and enters EL2 (Hypervisor mode)
arm_64bit=1

# Load our 64-bit kernel
kernel=kernel8.img

# Disable automatic command line generation - we'll parse device tree instead
disable_commandline_tags=1

# ============================================================================
# Memory Configuration
# ============================================================================

# GPU memory split
# Minimum for headless operation, increase if using graphics
# Pi 4/400: Can go as low as 16MB
# Pi 3: Minimum 64MB recommended for some operations
gpu_mem=16

# ============================================================================
# UART Configuration (for debugging)
# ============================================================================

# Enable UART for serial console
enable_uart=1

# Enable UART output during bootloader stage (very early boot messages)
uart_2ndstage=1

# Set UART baud rate (default is 115200)
core_freq=250  # Required for stable UART on some Pi models

# ============================================================================
# Device Tree Configuration
# ============================================================================

# Specify device tree blob to load
